func buildOpenAPIDoc(app *fiber.App) map[string]any {
	type opKey struct {
		method string
		path   string
	}
	type candidate struct {
		method string
		path   string
		params []string
	}
	var candidates []candidate
	seen := map[opKey]struct{}{}
	hasGET := map[string]bool{}

	var routes []fiber.Route
	if app != nil {
		routes = app.GetRoutes(true)
	}

	for _, r := range routes {
		method := strings.ToUpper(r.Method)
		if _, ok := httpMethods[method]; !ok {
			continue
		}
		if r.Path == "" || r.Path == "/*" || r.Path == "*" {
			continue
		}
		if len(r.Handlers) == 0 {
			continue
		}

		oaPath, params := fiberPathToOpenAPI(r.Path)
		if oaPath == "" {
			continue
		}
		key := opKey{method: method, path: oaPath}
		if _, dup := seen[key]; dup {
			continue
		}
		seen[key] = struct{}{}
		if method == fiber.MethodGet {
			hasGET[oaPath] = true
		}
		candidates = append(candidates, candidate{method: method, path: oaPath, params: params})
	}

	paths := map[string]map[string]any{}
	for _, c := range candidates {
		// Drop Fiber auto-HEAD clones of GET routes. Keep HEAD-only paths.
		if c.method == fiber.MethodHead {
			if hasGET[c.path] {
				continue
			}
		}

		op := map[string]any{}
		if len(c.params) > 0 {
			plist := make([]map[string]any, 0, len(c.params))
			for _, name := range c.params {
				plist = append(plist, map[string]any{
					"name":     name,
					"in":       "path",
					"required": true,
					"schema":   map[string]any{"type": "string"},
				})
			}
			op["parameters"] = plist
		}
		if openAPIPublicPath.MatchString(c.path) {
			op["security"] = []map[string][]string{}
		}
		op["responses"] = map[string]any{
			"200": map[string]any{"description": "OK"},
		}

		methods, ok := paths[c.path]
		if !ok {
			methods = map[string]any{}
			paths[c.path] = methods
		}
		methods[strings.ToLower(c.method)] = op
	}

	ordered := make([]string, 0, len(paths))
	for p := range paths {
		ordered = append(ordered, p)
	}
	sort.Strings(ordered)
	sortedPaths := make(map[string]any, len(paths))
	for _, p := range ordered {
		sortedPaths[p] = paths[p]
	}

	title := "api-db-go"
	if app != nil && app.Config().AppName != "" {
		title = app.Config().AppName
	}

	return map[string]any{
		"openapi": "3.1.0",
		"info": map[string]any{
			"title":   title,
			"version": "0.1.0",
		},
		"paths": sortedPaths,
		"components": map[string]any{
			"securitySchemes": map[string]any{
				"bearerAuth": map[string]any{
					"type":   "http",
					"scheme": "bearer",
				},
			},
		},
		"security": []map[string][]string{
			{"bearerAuth": {}},
		},
	}
}

func ParseBandwidthIngest(body []byte) (dbbandwidth.Batch, error) {
	var raw struct {
		Source          *string         `json:"source"`
		InstanceID      *string         `json:"instanceId"`
		BucketStartedAt *string         `json:"bucketStartedAt"`
		BucketEndedAt   *string         `json:"bucketEndedAt"`
		Entries         json.RawMessage `json:"entries"`
	}
	if err := json.Unmarshal(body, &raw); err != nil {
		return dbbandwidth.Batch{}, err
	}

	source, err := bwRequireStr(raw.Source, "source", 1, 64)
	if err != nil {
		return dbbandwidth.Batch{}, err
	}
	instanceID, err := bwRequireStr(raw.InstanceID, "instanceId", 1, 128)
	if err != nil {
		return dbbandwidth.Batch{}, err
	}
	startedAt, err := bwRequireDatetime(raw.BucketStartedAt, "bucketStartedAt")
	if err != nil {
		return dbbandwidth.Batch{}, err
	}
	endedAt, err := bwRequireDatetime(raw.BucketEndedAt, "bucketEndedAt")
	if err != nil {
		return dbbandwidth.Batch{}, err
	}

	if raw.Entries == nil {
		return dbbandwidth.Batch{}, fmt.Errorf("Invalid input: 'entries' is required")
	}
	if jsonKind(raw.Entries) != "array" {
		return dbbandwidth.Batch{}, fmt.Errorf("Invalid input: 'entries' must be an array")
	}
	var elems []json.RawMessage
	if err := json.Unmarshal(raw.Entries, &elems); err != nil {
		return dbbandwidth.Batch{}, err
	}
	if len(elems) > 10_000 {
		return dbbandwidth.Batch{}, fmt.Errorf("Too big: 'entries' must have <=10000 items")
	}

	entries := make([]dbbandwidth.Entry, 0, len(elems))
	for i, el := range elems {
		e, err := parseBandwidthEntry(el, i)
		if err != nil {
			return dbbandwidth.Batch{}, err
		}
		entries = append(entries, e)
	}

	return dbbandwidth.Batch{
		Source:          source,
		InstanceID:      instanceID,
		BucketStartedAt: startedAt,
		BucketEndedAt:   endedAt,
		Entries:         entries,
	}, nil
}

@ api-db-go internal/infrastructure/kpiscraper/scraper.go:226-284 func aggregate [file changed, re-anchored +0]
func aggregate(samples []sample) tickResult {
	var playing, paused, stopped int
	regions := map[string]int{}
	pingSum := map[string]float64{}
	pingCount := map[string]int{}

	for _, sm := range samples {
		p := sm.player
		switch {
		case p.Paused:
			paused++
		case p.Playing:
			playing++
		default:
			stopped++
		}

		if sm.endpoint == "" {
			continue
		}
		region := dbkpi.ExtractRegion(sm.endpoint)
		if region == "" {
			region = "unknown"
		}
		regions[region]++

		// Ping sólo de los que están sonando de verdad (un pausado o un detenido
		// reportan un RTT viejo) y sólo si es positivo: Lavalink usa -1 para «sin
		// conexión» y 0 para «aún sin ack del heartbeat».
		if !p.Playing || p.Paused {
			continue
		}
		if p.Ping == nil || *p.Ping <= 0 {
			continue
		}
		pingSum[region] += *p.Ping
		pingCount[region]++
	}

	pingByRegion := map[string]dbkpi.PingAgg{}
	for region, sum := range pingSum {
		count := pingCount[region]
		if count == 0 {
			continue
		}
		pingByRegion[region] = dbkpi.PingAgg{
			AvgMs:       math.Round((sum/float64(count))*10) / 10,
			SampleCount: count,
		}
	}

	return tickResult{
		Playing:      playing,
		Paused:       paused,
		Stopped:      stopped,
		Regions:      regions,
		PingByRegion: pingByRegion,
	}
}

func cutoverGuard(ctx context.Context, cfg config.Postgres) error {
	conn, err := pgx.Connect(ctx, cfg.DSN())
	if err != nil {
		return fmt.Errorf("cutover guard: connect: %w", err)
	}
	defer func() { _ = conn.Close(ctx) }()

	var hasSchemaMigrations bool
	if err := conn.QueryRow(ctx,
		`SELECT to_regclass('public.schema_migrations') IS NOT NULL`,
	).Scan(&hasSchemaMigrations); err != nil {
		return fmt.Errorf("cutover guard: probe schema_migrations: %w", err)
	}
	if hasSchemaMigrations {
		return nil // golang-migrate already owns this database
	}

	var schemaPopulated bool
	if err := conn.QueryRow(ctx,
		`SELECT to_regclass('public.recordings') IS NOT NULL`,
	).Scan(&schemaPopulated); err != nil {
		return fmt.Errorf("cutover guard: probe schema: %w", err)
	}
	if !schemaPopulated {
		return nil // truly fresh database — safe to run from scratch
	}

	// Populated but untracked → cutover. Enrich the message: note Node's
	// tracker if present, and whether the schema looks complete through the
	// final migration, so the operator can tell a full clone (safe to seal)
	// from a partial snapshot (must reconcile before forcing).
	origin := "outside golang-migrate"
	var hasPgMigrations bool
	if err := conn.QueryRow(ctx,
		`SELECT to_regclass('public.pgmigrations') IS NOT NULL`,
	).Scan(&hasPgMigrations); err == nil && hasPgMigrations {
		var rows int
		if conn.QueryRow(ctx, `SELECT count(*) FROM public.pgmigrations`).Scan(&rows) == nil {
			origin = fmt.Sprintf("by the Node api-db (%d rows in public.pgmigrations)", rows)
		} else {
			origin = "by the Node api-db (public.pgmigrations present)"
		}
	}

	var complete bool
	_ = conn.QueryRow(ctx,
		`SELECT to_regclass('music.track_counter') IS NOT NULL`,
	).Scan(&complete)
	remedy := "the schema looks INCOMPLETE (music.track_counter is missing) — likely a partial " +
		"or older snapshot; do NOT force, reconcile it first"
	if complete {
		remedy = fmt.Sprintf("if it is fully migrated, seal the state ONCE with `make pg-force V=%d`, "+
			"then re-run `make pg-migrate`", MaxMigrationVersion)
	}

	return fmt.Errorf(
		"cutover guard: target schema is already populated (public.recordings exists) but has no "+
			"schema_migrations table — this database was migrated %s. Running from scratch would "+
			"re-execute the whole series against a populated schema. %s",
		origin, remedy,
	)

	// document field.
	if g.Commands != nil {
		if err := putJSON(out, "commands", g.Commands); err != nil {
			return nil, err
		}
	}
	// `profile` (the bot's per-guild bio override — see dbguild.GuildProfile
	// for why avatar/banner don't live here, Discord does) is likewise only
	// written when the document carries it: an absent subdocument means
	// "never configured for this guild" (see dbguild.DBGuild.Profile), and a

	// "nested field", not one flat key per scalar. Payoff of that choice:
	// `GuildProfile` shrank from three fields (avatar/banner/bio) down to
	// one (bio) and needed ZERO changes here — json.Marshal(g.Profile) just
	// emits fewer keys, unlike a flat-keys scheme which would have needed
	// the `profile_avatar`/`profile_banner` lines deleted by hand.
	if g.Profile != nil {
		if err := putJSON(out, "profile", g.Profile); err != nil {
			return nil, err
		}
	}
	if err := putJSON(out, "module_music", g.ModuleMusic); err != nil {
		return nil, err
	}
	if err := putJSON(out, "module_greeter", g.ModuleGreeter); err != nil {
		return nil, err
	}
	if err := putJSON(out, "module_logs", g.ModuleLogs); err != nil {
		return nil, err
	}
	if err := putJSON(out, "module_modmail", g.ModuleModmail); err != nil {
		return nil, err
	}
	if err := putJSON(out, "module_tempvoice", g.ModuleTempvoice); err != nil {
		return nil, err
	}
	if err := putJSON(out, "module_giveaway", g.ModuleGiveaway); err != nil {
		return nil, err
	}
	if err := putJSON(out, "module_moderation", g.ModuleModeration); err != nil {
		return nil, err
	}
	if err := putJSON(out, "module_automation", g.ModuleAutomation); err != nil {
		return nil, err
	}

	out["createdAt"] = formatDate(g.CreatedAt.Time)
	out["updatedAt"] = formatDate(g.UpdatedAt.Time)

	return out, nil
}

		}

	}

		// Bring the live schema up to date in place. FT.ALTER is the only
		// non-destructive path; FT.DROPINDEX would force a full reindex.
		for _, f := range searchFields {
			args := append([]string{idxMembers, "SCHEMA", "ADD", f.name}, f.args...)
			if err := client.FtCommand(ctx, "FT.ALTER", args...); err != nil {
				if strings.Contains(strings.ToLower(err.Error()), "duplicate field") {
					continue
				}
				log.Error().Str("error", err.Error()).Str("field", f.name).
					Msg("FT.ALTER idx_members field failed")
				continue
			}
			log.Info().Str("field", f.name).Msg("FT.ALTER idx_members added field")
		}
		return
	}

	// Missing index → create it whole. Schema: guildId TAG SORTABLE, id TAG
	// SORTABLE, then the three `_lc` TEXT NOSTEM fields. FILTER-scoped to
	// `@_type=='member'` so channel/role hashes under the shared `guilds:`
	// prefix are excluded.

		schema = append(schema, f.name)
		schema = append(schema, f.args...)
	}

func LoadSecrets() (configured bool, err error) {
	if !isConfigured() {
		return false, nil
	}
	url := os.Getenv("INFISICAL_URL")
	clientID := os.Getenv("INFISICAL_CLIENT_ID")
	clientSecret := os.Getenv("INFISICAL_CLIENT_SECRET")
	environment := os.Getenv("INFISICAL_ENVIRONMENT")

	ctx, cancel := context.WithTimeout(context.Background(), 5*time.Minute)
	defer cancel()

	err = withRetry(ctx, func() error {
		client, authErr := newInfisicalClient(ctx, url, clientID, clientSecret)
		if authErr != nil {
			return authErr
		}
		if err := fetchProject(client, projectCommon, environment); err != nil {
			return fmt.Errorf("common: %w", err)
		}
		if err := fetchProject(client, projectAPIDBGo, environment); err != nil {
			return fmt.Errorf("api-db-go: %w", err)
		}
		return nil
	})

	if err != nil {
		return true, fmt.Errorf("infisical secrets: %w", err)
	}
	return true, nil
}

export class RecommendationsCache extends BaseCache {
	private cacheKey(isrc: string, storefront: string): string {
		return `recommendations:${isrc.toLowerCase()}:${storefront.toLowerCase()}`;
	}

	async getResults(isrc: string, storefront: string): Promise<ChipbotRecommendationsResponse | null> {
		return this.get<ChipbotRecommendationsResponse>(this.cacheKey(isrc, storefront));
	}

	async setResults(isrc: string, storefront: string, data: ChipbotRecommendationsResponse): Promise<void> {
		await this.set(this.cacheKey(isrc, storefront), data, TTL_5_MIN);
	}
}
